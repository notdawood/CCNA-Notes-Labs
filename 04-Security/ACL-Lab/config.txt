========================
🔹 Router R1 Config
===================

enable
configure terminal

hostname Router1

interface g0/0
ip address 192.168.10.1 255.255.255.0
no shutdown

interface g0/1
ip address 192.168.20.1 255.255.255.0
no shutdown

access-list 1 deny host 192.168.10.10
access-list 1 permit any

interface g0/1
ip access-group 1 out

end
write
